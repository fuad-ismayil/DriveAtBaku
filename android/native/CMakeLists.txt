cmake_minimum_required(VERSION 3.22)
project(DriveAtBakuNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOT_CPP_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/android/godot-cpp" CACHE PATH "Pinned godot-cpp checkout")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
target_compile_options(godot-cpp PRIVATE $<$<CXX_COMPILER_ID:Clang,GNU>:-include;cstdlib>)
add_library(driveatbaku SHARED extension.cpp third_party/fdlibm/ieee754.cpp)
target_link_libraries(driveatbaku PRIVATE godot-cpp)
target_compile_options(driveatbaku PRIVATE $<$<CXX_COMPILER_ID:Clang,GNU>:-ffp-contract=off>)
set_target_properties(driveatbaku PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../game/native" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../game/native")
if(ANDROID)
  set_target_properties(driveatbaku PROPERTIES OUTPUT_NAME "libdriveatbaku.android.arm64")
else()
  set_target_properties(driveatbaku PROPERTIES OUTPUT_NAME "driveatbaku.windows.x86_64")
  target_link_options(driveatbaku PRIVATE -static)
endif()
